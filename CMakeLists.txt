cmake_minimum_required(VERSION 3.16)
project(TSPR3 VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
include(CMakePackageConfigHelpers)
option(TSPR3_BUILD_EXAMPLES "Build C/C++ examples" ON)
add_library(tspr3 STATIC src/c/tspr3_tri.c src/c/tspr3_profile.c src/c/tspr3_log_selector.c src/c/tspr3_library.c)
add_library(TSPR3::tspr3 ALIAS tspr3)
set_target_properties(tspr3 PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_include_directories(tspr3 PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src/c/internal)
if(MSVC)
    target_compile_options(tspr3 PRIVATE /W4 /WX)
else()
    target_compile_options(tspr3 PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(tspr3 PUBLIC m)
endif()
if(TSPR3_BUILD_EXAMPLES)
    add_executable(tspr3_example_c examples/c/basic.c)
    add_executable(tspr3_example_cpp examples/cpp/basic.cpp)
    target_link_libraries(tspr3_example_c PRIVATE TSPR3::tspr3)
    target_link_libraries(tspr3_example_cpp PRIVATE TSPR3::tspr3)
    target_compile_features(tspr3_example_cpp PRIVATE cxx_std_17)
endif()
if(BUILD_TESTING)
    add_executable(tspr3_library_test_c tests/c/library_test.c)
    add_executable(tspr3_library_test_cpp tests/cpp/library_test.cpp)
    target_link_libraries(tspr3_library_test_c PRIVATE TSPR3::tspr3)
    target_link_libraries(tspr3_library_test_cpp PRIVATE TSPR3::tspr3)
    target_compile_features(tspr3_library_test_cpp PRIVATE cxx_std_17)
    add_test(NAME library_c COMMAND tspr3_library_test_c)
    add_test(NAME library_cpp COMMAND tspr3_library_test_cpp)
    add_executable(tspr3_core_test tests/c/tri_test.c)
    add_executable(tspr3_profile_test tests/c/profile_test.c)
    foreach(test_target tspr3_core_test tspr3_profile_test)
        target_link_libraries(${test_target} PRIVATE TSPR3::tspr3)
        target_include_directories(${test_target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src/c/internal)
        set_target_properties(${test_target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES)
    endforeach()
    add_test(NAME core_strategies COMMAND tspr3_core_test)
    add_test(NAME selector_policy COMMAND tspr3_profile_test)
endif()
install(TARGETS tspr3 EXPORT TSPR3Targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT TSPR3Targets NAMESPACE TSPR3:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TSPR3)
configure_package_config_file(cmake/TSPR3Config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/TSPR3Config.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TSPR3)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/TSPR3ConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/TSPR3Config.cmake ${CMAKE_CURRENT_BINARY_DIR}/TSPR3ConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TSPR3)
install(FILES docs/LIBRARY.md docs/ARCHITECTURE.md docs/VALIDATION.md models/c/log_model.json
    DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/TSPR3)
install(FILES LICENSE NOTICE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/TSPR3)
install(FILES docs/research/RUN-2M.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/TSPR3/research)
install(FILES examples/c/basic.c examples/cpp/basic.cpp DESTINATION ${CMAKE_INSTALL_DATADIR}/TSPR3/examples)
